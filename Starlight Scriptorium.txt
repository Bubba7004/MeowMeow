Shinies
	*wanderingIdea
		on click:log A thought drifts by...
		movement:onRight moveLeft fade bounce:0.05
		frequency:45
		frequency variation:20
		icon:stuff/wanderingIdea.png
		class:bigButton
		on click:
			$boost=1
			if (have brainstorm) $boost=3
			if (chance(30%))
				$amount=max(15,random(words:ps*60*2,words:ps*60*5))*$boost
				toast The idea blooms into<//><b>[$amount] words</b>!
				yield $amount words
			else
				$amount=1*$boost
				toast The idea crystallizes into<//><b>[$amount] starlight[s?$amount]</b>!
				yield $amount starlight
			end
		end
